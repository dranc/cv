{{- $start := time .start -}} {{- $now := now -}} {{- $end := time (default $now .end) -}} {{- $isCurrentMonth := eq (time.Format "2006-01" $now) (time.Format "2006-01" $end) -}} {{- $endStr := cond $isCurrentMonth (i18n "now") (time.Format "Jan 2006" $end) -}} {{- $startStr := time.Format (cond (eq $start.Year $end.Year) "Jan" "Jan 2006") $start -}} {{- $elapsedMonths := div (sub $end.Unix $start.Unix) (mul 60 60 24 30) -}} {{- $years := div $elapsedMonths 12 -}} {{- $months := mod $elapsedMonths 12 -}}

{{ i18n "date_to_date" (dict "start" $startStr "end" $endStr) }}

{{- if gt $years 0 -}} {{- $title := printf "%s %s" (i18n "years_count" (dict "Count" $years)) (i18n "months_count" $months) -}} {{- $half := false -}} {{- if ge $months 10 -}} {{- $years = add $years 1 -}} {{- else if ge $months 4 -}} {{- $half = true -}} {{- end -}}

{{ i18n "years_count" (dict "Count" $years "Half" $half) }}

{{- else -}}

{{ i18n "months_count" $months }}

{{- end -}}
